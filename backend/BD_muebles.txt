CREATE DATABASE IF NOT EXISTS tienda_muebles;
USE tienda_muebles;

CREATE TABLE administrador (
    id_admin INT AUTO_INCREMENT PRIMARY KEY,
    nombre_usuario VARCHAR(100) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL
);

CREATE TABLE inventario (
    id_inventario INT AUTO_INCREMENT PRIMARY KEY,
    fecha_ultima_actualizacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    id_admin INT
);

CREATE TABLE categorias (
    id_categoria INT AUTO_INCREMENT PRIMARY KEY,
    nombre_categoria VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE muebles (
    id_mueble INT AUTO_INCREMENT PRIMARY KEY,
    sku VARCHAR(50) UNIQUE NOT NULL,
    nombre VARCHAR(150) NOT NULL,
    imagen TEXT,
    medidas VARCHAR(255),       
    precio_costo INT NOT NULL, 
    precio_venta INT,          
    peso_kg DECIMAL(5,2),       
    bultos INT DEFAULT 1,
    observaciones VARCHAR(255),
    id_categoria INT
);

CREATE TABLE solicitudes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre_cliente VARCHAR(255) NOT NULL,
    telefono VARCHAR(100) NOT NULL,
    direccion VARCHAR(255),
    metodo_pago VARCHAR(50),
    mensaje TEXT,
    fecha DATETIME DEFAULT CURRENT_TIMESTAMP,
    estado VARCHAR(50) DEFAULT 'PENDIENTE',
    producto_id INT
);

CREATE TABLE calculos_medidas (
    id_calculo INT AUTO_INCREMENT PRIMARY KEY,
    descripcion_mueble VARCHAR(150) NOT NULL,
    alto_base DECIMAL(6,2) NOT NULL,
    ancho_base DECIMAL(6,2) NOT NULL,
    profundidad_base DECIMAL(6,2) NOT NULL,
    grosor_melamina_mm DECIMAL(4,2) NOT NULL,
    fecha_calculo TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    id_admin INT
);

CREATE TABLE detalle_calculo (
    id_detalle INT AUTO_INCREMENT PRIMARY KEY,
    id_calculo INT,
    nombre_pieza VARCHAR(100) NOT NULL,
    cantidad INT NOT NULL,
    largo_mm DECIMAL(6,1) NOT NULL,
    ancho_mm DECIMAL(6,1) NOT NULL
);


ALTER TABLE inventario
ADD CONSTRAINT fk_inventario_admin 
FOREIGN KEY (id_admin) REFERENCES administrador(id_admin) ON DELETE SET NULL;

ALTER TABLE muebles
ADD CONSTRAINT fk_muebles_categoria 
FOREIGN KEY (id_categoria) REFERENCES categorias(id_categoria) ON DELETE SET NULL;

ALTER TABLE solicitudes
ADD CONSTRAINT fk_solicitudes_mueble 
FOREIGN KEY (producto_id) REFERENCES muebles(id_mueble) ON DELETE SET NULL;

ALTER TABLE calculos_medidas
ADD CONSTRAINT fk_calculos_admin 
FOREIGN KEY (id_admin) REFERENCES administrador(id_admin) ON DELETE SET NULL;

ALTER TABLE detalle_calculo
ADD CONSTRAINT fk_detalle_calculo 
FOREIGN KEY (id_calculo) REFERENCES calculos_medidas(id_calculo) ON DELETE CASCADE;

-- ==========================================
-- TABLAS PARA MUEBLES PROPIOS (CRUD ADMIN)
-- ==========================================

CREATE TABLE categorias_propias (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100) NOT NULL UNIQUE,
    orden INT DEFAULT 0,
    activa BOOLEAN DEFAULT TRUE
);

CREATE TABLE muebles_propios (
    id INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(150) NOT NULL,
    descripcion TEXT,
    video_url VARCHAR(500),
    medidas VARCHAR(255),
    precio INT NOT NULL,
    categoria_id INT NOT NULL,
    activo BOOLEAN DEFAULT TRUE,
    fecha_creacion TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (categoria_id) REFERENCES categorias_propias(id) ON UPDATE CASCADE ON DELETE RESTRICT
);

CREATE TABLE muebles_propios_imagenes (
    id INT AUTO_INCREMENT PRIMARY KEY,
    mueble_id INT NOT NULL,
    imagen_url VARCHAR(255) NOT NULL,
    orden INT DEFAULT 0,
    FOREIGN KEY (mueble_id) REFERENCES muebles_propios(id) ON DELETE CASCADE
);
